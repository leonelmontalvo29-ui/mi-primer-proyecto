<script>
  var _creditos = [];
  var _clientes = [];

  document.addEventListener('DOMContentLoaded', function () {
    cargarCreditos();
    cargarClientes();
    cargarPersonalidades();
    cargarDashboard();
    document.querySelectorAll('.modal-overlay').forEach(function (ov) {
      ov.addEventListener('click', function (e) { if (e.target === ov) ov.classList.remove('abierto'); });
    });
  });

  // ---------- Navegación ----------
  function cambiarTab(tab) {
    document.querySelectorAll('.tab-panel').forEach(function (p) { p.classList.remove('activo'); });
    document.querySelectorAll('.tab-btn').forEach(function (b) { b.classList.remove('active'); });
    document.getElementById('tab-' + tab).classList.add('activo');
    document.querySelector('.tab-btn[data-tab="' + tab + '"]').classList.add('active');
    if (tab === 'resumen') resumenPopularClientes();
  }

  // ---------- Utilidades ----------
  function esc(v) {
    return String(v == null ? '' : v)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  }
  function fechaDisplay(iso) {
    if (!iso) return '';
    var p = String(iso).split('-');
    return p.length === 3 ? (p[2] + '/' + p[1] + '/' + p[0]) : iso;
  }
  function toast(msg, esError) {
    var t = document.getElementById('toast');
    t.textContent = msg;
    t.className = 'toast visible' + (esError ? ' error' : '');
    setTimeout(function () { t.className = 'toast'; }, 4500);
  }
  function abrirModal(id) { document.getElementById(id).classList.add('abierto'); }
  function cerrarModal(id) { document.getElementById(id).classList.remove('abierto'); }

  function opcionesClientes(selId) {
    var sel = document.getElementById(selId);
    if (!sel) return;
    sel.innerHTML = '<option value="">Selecciona…</option>' +
      _clientes.map(function (c) { return '<option value="' + esc(c['ID_Cliente']) + '">' + esc(c['Nombre del cliente']) + '</option>'; }).join('');
  }

  // ---------- Buscadores ----------
  function filtrarTabla(inputId, tablaId) {
    var q = (document.getElementById(inputId).value || '').toLowerCase();
    var t = document.getElementById(tablaId); if (!t) return;
    var filas = t.querySelectorAll('tbody tr');
    for (var i = 0; i < filas.length; i++) {
      var tr = filas[i];
      if (tr.classList.contains('grupo-credito')) { tr.style.display = ''; continue; }
      tr.style.display = tr.textContent.toLowerCase().indexOf(q) !== -1 ? '' : 'none';
    }
  }
  function filtrarTablaContenedor(inputId, contId) {
    var q = (document.getElementById(inputId).value || '').toLowerCase();
    var cont = document.getElementById(contId); if (!cont) return;
    var filas = cont.querySelectorAll('table tbody tr');
    for (var i = 0; i < filas.length; i++) {
      filas[i].style.display = filas[i].textContent.toLowerCase().indexOf(q) !== -1 ? '' : 'none';
    }
  }
  function filtrarChecklist() {
    var q = (document.getElementById('buscar-checklist').value || '').toLowerCase();
    var cats = document.querySelectorAll('#ck-contenido .ck-cat');
    for (var c = 0; c < cats.length; c++) {
      var filas = cats[c].querySelectorAll('tbody tr'), visibles = 0;
      for (var i = 0; i < filas.length; i++) {
        var ok = filas[i].textContent.toLowerCase().indexOf(q) !== -1;
        filas[i].style.display = ok ? '' : 'none';
        if (ok) visibles++;
      }
      cats[c].style.display = (q && visibles === 0) ? 'none' : '';
    }
  }

  // =========================================================
  //  CLIENTES
  // =========================================================
  function cargarClientes() {
    google.script.run
      .withSuccessHandler(function (cli) { _clientes = cli || []; pintarClientes(); })
      .withFailureHandler(function (err) { toast('Error al cargar clientes: ' + err.message, true); })
      .obtenerClientes();
  }
  function pintarClientes() {
    var tbody = document.querySelector('#tabla-clientes tbody');
    if (!tbody) return;
    if (_clientes.length === 0) { tbody.innerHTML = '<tr><td colspan="3" class="vacio">Aún no hay clientes registrados.</td></tr>'; return; }
    tbody.innerHTML = _clientes.map(function (c) {
      return '<tr>' +
        '<td>' + esc(c['ID_Cliente']) + '</td>' +
        '<td>' + esc(c['Nombre del cliente']) + '</td>' +
        '<td><div class="acciones">' +
          '<button class="btn btn-ghost btn-sm" onclick="editarCliente(' + c['ID_Cliente'] + ')">Editar</button>' +
          '<button class="btn btn-warn btn-sm" onclick="eliminarCliente(' + c['ID_Cliente'] + ')">Eliminar</button>' +
        '</div></td>' +
      '</tr>';
    }).join('');
  }
  function abrirFormCliente() {
    document.getElementById('titulo-modal-cliente').textContent = 'Nuevo cliente';
    document.getElementById('cliente-id').value = '';
    document.getElementById('cliente-nombre').value = '';
    abrirModal('modal-cliente');
  }
  function editarCliente(id) {
    var c = _clientes.filter(function (x) { return String(x['ID_Cliente']) === String(id); })[0];
    if (!c) return;
    document.getElementById('titulo-modal-cliente').textContent = 'Editar cliente';
    document.getElementById('cliente-id').value = c['ID_Cliente'];
    document.getElementById('cliente-nombre').value = c['Nombre del cliente'] || '';
    abrirModal('modal-cliente');
  }
  function guardarCliente() {
    var id = document.getElementById('cliente-id').value;
    var nombre = document.getElementById('cliente-nombre').value.trim();
    if (!nombre) { toast('El nombre del cliente es obligatorio.', true); return; }
    var btn = document.getElementById('btn-guardar-cliente');
    btn.disabled = true; btn.textContent = 'Guardando…';
    var handler = google.script.run
      .withSuccessHandler(function () { btn.disabled = false; btn.textContent = 'Guardar'; cerrarModal('modal-cliente'); toast(id ? 'Cliente actualizado.' : 'Cliente creado.'); cargarClientes(); cargarCreditos(); })
      .withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); });
    if (id) handler.actualizarCliente(id, nombre); else handler.crearCliente(nombre);
  }
  function eliminarCliente(id) {
    var c = _clientes.filter(function (x) { return String(x['ID_Cliente']) === String(id); })[0];
    var nombre = c ? c['Nombre del cliente'] : ('#' + id);
    if (!confirm('¿Eliminar al cliente "' + nombre + '"?\n\nSi no tiene créditos activos, se eliminarán también sus personalidades y checklists asociados.')) return;
    google.script.run
      .withSuccessHandler(function (res) {
        if (res && res.ok) { toast('Cliente eliminado (' + res.personalidadesBorradas + ' personalidad(es), ' + res.checklistBorrados + ' fila(s) de checklist).'); cargarClientes(); cargarPersonalidades(); }
        else { toast((res && res.mensaje) ? res.mensaje : 'No se pudo eliminar.', true); }
      })
      .withFailureHandler(function (err) { toast('Error: ' + err.message, true); })
      .eliminarCliente(id);
  }

  // =========================================================
  //  CRÉDITOS
  // =========================================================
  function cargarCreditos() {
    google.script.run
      .withSuccessHandler(pintarCreditos)
      .withFailureHandler(function (err) { toast('Error al cargar créditos: ' + err.message, true); })
      .obtenerCreditos();
  }
  function pintarCreditos(creditos) {
    _creditos = creditos || [];
    var tbody = document.querySelector('#tabla-creditos tbody');
    if (_creditos.length === 0) { tbody.innerHTML = '<tr><td colspan="6" class="vacio">Aún no hay créditos registrados.</td></tr>'; return; }
    tbody.innerHTML = _creditos.map(function (c) {
      var activo = String(c['Estado']).toLowerCase() === 'activo';
      var badge = '<span class="estado ' + (activo ? 'estado-activo' : 'estado-inactivo') + '">' + esc(c['Estado']) + '</span>';
      var btnDesactivar = activo ? '<button class="btn btn-warn btn-sm" onclick="desactivarCredito(' + c['ID'] + ')">Desactivar</button>' : '';
      return '<tr>' +
        '<td>' + esc(c['ID']) + '</td>' +
        '<td>' + esc(c['Nombre del crédito']) + '</td>' +
        '<td>' + esc(c['Cliente']) + '</td>' +
        '<td>' + esc(fechaDisplay(c['Fecha de inicio'])) + '</td>' +
        '<td>' + badge + '</td>' +
        '<td><div class="acciones">' +
          '<button class="btn btn-ghost btn-sm" onclick="editarCredito(' + c['ID'] + ')">Editar</button>' + btnDesactivar +
        '</div></td>' +
      '</tr>';
    }).join('');
  }
  function abrirFormCredito() {
    if (_clientes.length === 0) { toast('Primero registra al menos un cliente.', true); return; }
    document.getElementById('titulo-modal-credito').textContent = 'Nuevo crédito';
    document.getElementById('credito-id').value = '';
    document.getElementById('lbl-credito-cliente').style.display = '';
    document.getElementById('nota-credito').style.display = '';
    opcionesClientes('credito-cliente');
    document.getElementById('credito-cliente').disabled = false;
    document.getElementById('credito-nombre').value = '';
    document.getElementById('credito-fecha').value = '';
    abrirModal('modal-credito');
  }
  function editarCredito(id) {
    var c = _creditos.filter(function (x) { return String(x['ID']) === String(id); })[0];
    if (!c) return;
    document.getElementById('titulo-modal-credito').textContent = 'Editar crédito';
    document.getElementById('credito-id').value = c['ID'];
    document.getElementById('lbl-credito-cliente').style.display = 'none';
    document.getElementById('nota-credito').style.display = 'none';
    document.getElementById('credito-nombre').value = c['Nombre del crédito'] || '';
    document.getElementById('credito-fecha').value = c['Fecha de inicio'] || '';
    abrirModal('modal-credito');
  }
  function guardarCredito() {
    var id = document.getElementById('credito-id').value;
    var btn = document.getElementById('btn-guardar-credito');
    if (id) {
      var datosU = { nombre: document.getElementById('credito-nombre').value.trim(), fechaInicio: document.getElementById('credito-fecha').value };
      if (!datosU.nombre) { toast('El nombre del crédito es obligatorio.', true); return; }
      btn.disabled = true; btn.textContent = 'Guardando…';
      google.script.run
        .withSuccessHandler(function () { btn.disabled = false; btn.textContent = 'Guardar'; cerrarModal('modal-credito'); toast('Crédito actualizado.'); cargarCreditos(); })
        .withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); })
        .actualizarCredito(id, datosU);
      return;
    }
    var datos = { idCliente: document.getElementById('credito-cliente').value, nombre: document.getElementById('credito-nombre').value.trim(), fechaInicio: document.getElementById('credito-fecha').value };
    if (!datos.idCliente || !datos.nombre) { toast('Cliente y Nombre son obligatorios.', true); return; }
    btn.disabled = true; btn.textContent = 'Guardando…';
    google.script.run
      .withSuccessHandler(function (res) { btn.disabled = false; btn.textContent = 'Guardar'; cerrarModal('modal-credito'); toast('Crédito creado. Personalidades heredadas: ' + res.personalidadesHeredadas + '.'); cargarCreditos(); })
      .withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); })
      .crearCredito(datos);
  }
  function desactivarCredito(id) {
    if (!confirm('¿Desactivar este crédito? Su estado pasará a "Inactivo".')) return;
    google.script.run
      .withSuccessHandler(function () { toast('Crédito desactivado.'); cargarCreditos(); })
      .withFailureHandler(function (err) { toast('Error: ' + err.message, true); })
      .desactivarCredito(id);
  }

  // =========================================================
  //  PERSONALIDADES
  // =========================================================
  var _personalidades = [];
  function cargarPersonalidades() {
    google.script.run
      .withSuccessHandler(pintarPersonalidades)
      .withFailureHandler(function (err) { toast('Error al cargar personalidades: ' + err.message, true); })
      .obtenerPersonalidades();
  }
  function pintarPersonalidades(personalidades) {
    _personalidades = personalidades || [];
    var tbody = document.querySelector('#tabla-personalidades tbody');
    if (_personalidades.length === 0) { tbody.innerHTML = '<tr><td colspan="4" class="vacio">Aún no hay personalidades registradas.</td></tr>'; return; }
    var html = '';
    var clienteActual = null;
    _personalidades.forEach(function (p) {
      var cli = String(p['Cliente']);
      if (cli !== clienteActual) { clienteActual = cli; html += '<tr class="grupo-credito"><td colspan="4">👥 ' + esc(cli) + '</td></tr>'; }
      html += '<tr>' +
        '<td>' + esc(p['ID']) + '</td>' +
        '<td>' + esc(p['Nombre']) + (p['Vinculo'] ? ' <small style="color:#8a6d00">— ' + esc(p['Vinculo']) + '</small>' : '') + '</td>' +
        '<td>' + esc(p['Tipo']) + '</td>' +
        '<td><div class="acciones">' +
          '<button class="btn btn-ghost btn-sm" onclick="editarPersonalidad(' + p['ID'] + ')">Editar</button>' +
          '<button class="btn btn-warn btn-sm" onclick="eliminarPersonalidad(' + p['ID'] + ')">Eliminar</button>' +
        '</div></td>' +
      '</tr>';
    });
    tbody.innerHTML = html;
  }
  function fmtTiposSeleccionados() {
    return Array.prototype.slice.call(document.querySelectorAll('#personalidad-tipos .pt-chk'))
      .filter(function (c) { return c.checked; }).map(function (c) { return c.value; });
  }
  function fmtSetTipos(tipoStr) {
    var sel = String(tipoStr || '').split(',').map(function (t) { return t.trim().toLowerCase(); });
    document.querySelectorAll('#personalidad-tipos .pt-chk').forEach(function (c) {
      c.checked = sel.indexOf(c.value.toLowerCase()) !== -1;
    });
  }
  function pPopularPadres(idCliente, excluirId, seleccionado) {
    var sel = document.getElementById('personalidad-padre');
    var opts = (_personalidades || []).filter(function (p) {
      return String(p['ID_Cliente']) === String(idCliente) && String(p['ID']) !== String(excluirId || '');
    });
    sel.innerHTML = '<option value="">— Directo del acreditado —</option>' +
      opts.map(function (p) { return '<option value="' + esc(p['ID']) + '">' + esc(p['Nombre']) + ' (' + esc(p['Tipo']) + ')</option>'; }).join('');
    if (seleccionado) sel.value = String(seleccionado);
  }
  function abrirFormPersonalidad() {
    if (_clientes.length === 0) { toast('Primero registra al menos un cliente.', true); return; }
    document.getElementById('titulo-modal-personalidad').textContent = 'Nueva personalidad';
    document.getElementById('personalidad-id').value = '';
    document.getElementById('lbl-personalidad-cliente').style.display = '';
    document.getElementById('nota-personalidad').style.display = '';
    document.getElementById('nota-personalidad').textContent = 'Marca todos los roles que tenga esta persona. Los documentos se combinan sin repetirse.';
    opcionesClientes('personalidad-cliente');
    var selCli = document.getElementById('personalidad-cliente');
    selCli.disabled = false; selCli.value = '';
    selCli.onchange = function () { pPopularPadres(selCli.value, '', ''); };
    document.getElementById('personalidad-nombre').value = '';
    fmtSetTipos('');
    pPopularPadres('', '', '');
    abrirModal('modal-personalidad');
  }
  function editarPersonalidad(id) {
    var p = _personalidades.filter(function (x) { return String(x['ID']) === String(id); })[0];
    if (!p) return;
    document.getElementById('titulo-modal-personalidad').textContent = 'Editar personalidad';
    document.getElementById('personalidad-id').value = p['ID'];
    document.getElementById('lbl-personalidad-cliente').style.display = 'none';
    document.getElementById('nota-personalidad').style.display = '';
    document.getElementById('nota-personalidad').textContent = 'Si agregas roles, se añadirán solo los documentos que falten. Nunca se borra lo ya registrado.';
    document.getElementById('personalidad-nombre').value = p['Nombre'] || '';
    fmtSetTipos(p['Tipo']);
    pPopularPadres(p['ID_Cliente'], p['ID'], p['PerteneceA']);
    abrirModal('modal-personalidad');
  }
  function guardarPersonalidad() {
    var id = document.getElementById('personalidad-id').value;
    var nombre = document.getElementById('personalidad-nombre').value.trim();
    var tipos = fmtTiposSeleccionados();
    var padre = document.getElementById('personalidad-padre').value;
    var btn = document.getElementById('btn-guardar-personalidad');
    if (!nombre || tipos.length === 0) { toast('Nombre y al menos un Tipo son obligatorios.', true); return; }
    var tipo = tipos.join(', ');
    btn.disabled = true; btn.textContent = 'Guardando…';
    if (id) {
      google.script.run
        .withSuccessHandler(function (res) { btn.disabled = false; btn.textContent = 'Guardar'; cerrarModal('modal-personalidad'); toast('Personalidad actualizada. Documentos agregados: ' + res.documentosAgregados + '.'); cargarPersonalidades(); })
        .withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); })
        .actualizarPersonalidad(id, { nombre: nombre, tipo: tipo, perteneceA: padre });
      return;
    }
    var datos = { idCliente: document.getElementById('personalidad-cliente').value, nombre: nombre, tipo: tipo, perteneceA: padre };
    if (!datos.idCliente) { toast('Selecciona un cliente.', true); return; }
    google.script.run
      .withSuccessHandler(function (res) { btn.disabled = false; btn.textContent = 'Guardar'; cerrarModal('modal-personalidad'); toast('Personalidad creada. Documentos generados en ' + res.creditos + ' crédito(s).'); cargarPersonalidades(); })
      .withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); })
      .crearPersonalidad(datos);
  }
  function eliminarPersonalidad(id) {
    var p = _personalidades.filter(function (x) { return String(x['ID']) === String(id); })[0];
    var nombre = p ? p['Nombre'] : ('#' + id);
    if (!confirm('¿Eliminar la personalidad "' + nombre + '"?\n\nSe borrarán también sus filas del checklist en todos los créditos.')) return;
    google.script.run
      .withSuccessHandler(function (res) { toast('Personalidad eliminada. Se borraron ' + res.checklistBorrados + ' fila(s) del checklist.'); cargarPersonalidades(); })
      .withFailureHandler(function (err) { toast('Error: ' + err.message, true); })
      .eliminarPersonalidad(id);
  }

  // =========================================================
  //  CHECKLIST
  // =========================================================
  var _ckClienteId = '', _ckCreditoId = '', _ckPersonalidadId = '', _ckItems = [];

  function ckPopularClientes() {
    var sel = document.getElementById('ck-cliente');
    var actual = sel.value;
    opcionesClientes('ck-cliente');
    if (actual) sel.value = actual;
  }
  function ckLimpiarContenido(msg) {
    document.getElementById('ck-progreso').style.display = 'none';
    document.getElementById('ck-contenido').innerHTML = '<p class="vacio">' + msg + '</p>';
  }
  function ckSeleccionarCliente() {
    _ckClienteId = document.getElementById('ck-cliente').value;
    _ckCreditoId = ''; _ckPersonalidadId = '';
    var selC = document.getElementById('ck-credito');
    var selP = document.getElementById('ck-personalidad');
    selC.value = ''; selP.value = '';
    ckLimpiarContenido('Selecciona un crédito y una personalidad para ver el checklist.');
    if (!_ckClienteId) {
      selC.disabled = true; selC.innerHTML = '<option value="">Selecciona un cliente primero…</option>';
      selP.disabled = true; selP.innerHTML = '<option value="">Selecciona un crédito primero…</option>';
      return;
    }
    selC.disabled = true; selC.innerHTML = '<option value="">Cargando…</option>';
    google.script.run
      .withSuccessHandler(function (creditos) {
        if (!creditos || creditos.length === 0) { selC.innerHTML = '<option value="">(Sin créditos)</option>'; selC.disabled = true; }
        else { selC.innerHTML = '<option value="">Selecciona…</option>' + creditos.map(function (c) { return '<option value="' + esc(c['ID']) + '">' + esc(c['Nombre del crédito']) + ' (' + esc(c['Estado']) + ')</option>'; }).join(''); selC.disabled = false; }
      })
      .withFailureHandler(function (err) { selC.innerHTML = '<option value="">Error</option>'; toast('Error: ' + err.message, true); })
      .obtenerCreditosPorCliente(_ckClienteId, false);
    selP.disabled = true; selP.innerHTML = '<option value="">Cargando…</option>';
    google.script.run
      .withSuccessHandler(function (personalidades) {
        if (!personalidades || personalidades.length === 0) { selP.innerHTML = '<option value="">(Este cliente no tiene personalidades)</option>'; selP.disabled = true; }
        else { selP.innerHTML = '<option value="">Selecciona…</option>' + personalidades.map(function (p) { return '<option value="' + esc(p['ID']) + '">' + esc(p['Etiqueta']) + ' (' + esc(p['Tipo']) + ')</option>'; }).join(''); selP.disabled = false; }
      })
      .withFailureHandler(function (err) { selP.innerHTML = '<option value="">Error</option>'; toast('Error: ' + err.message, true); })
      .obtenerPersonalidadesPorCliente(_ckClienteId);
  }
  function ckSeleccionarCredito() {
    _ckCreditoId = document.getElementById('ck-credito').value;
    if (!_ckCreditoId) { ckLimpiarContenido('Selecciona un crédito y una personalidad para ver el checklist.'); return; }
    if (_ckPersonalidadId) ckCargarChecklist(); else ckCargarAutorizacion();
  }
  function ckSeleccionarPersonalidad() {
    _ckPersonalidadId = document.getElementById('ck-personalidad').value;
    if (!_ckCreditoId) { ckLimpiarContenido('Selecciona un crédito.'); return; }
    if (_ckPersonalidadId) ckCargarChecklist(); else ckCargarAutorizacion();
  }
  function ckCargarAutorizacion() {
    document.getElementById('ck-progreso').style.display = 'none';
    document.getElementById('ck-contenido').innerHTML = '<p class="cargando">Cargando documentos del crédito…</p>';
    google.script.run
      .withSuccessHandler(function (res) { _ckItems = (res && res.autorizacion) ? res.autorizacion : []; ckRenderizar(); })
      .withFailureHandler(function (err) { document.getElementById('ck-contenido').innerHTML = '<p class="vacio">Error al cargar.</p>'; toast('Error: ' + err.message, true); })
      .obtenerChecklistCredito(_ckCreditoId);
  }
  function ckCargarChecklist() {
    document.getElementById('ck-progreso').style.display = 'none';
    document.getElementById('ck-contenido').innerHTML = '<p class="cargando">Cargando checklist…</p>';
    google.script.run
      .withSuccessHandler(function (res) {
        var aut = (res && res.autorizacion) ? res.autorizacion : [];
        var items = (res && res.items) ? res.items : [];
        _ckItems = aut.concat(items);
        ckRenderizar();
      })
      .withFailureHandler(function (err) { document.getElementById('ck-contenido').innerHTML = '<p class="vacio">Error al cargar el checklist.</p>'; toast('Error: ' + err.message, true); })
      .obtenerChecklist(_ckCreditoId, _ckPersonalidadId);
  }
  function ckRenderizar() {
    var cont = document.getElementById('ck-contenido');
    if (!_ckItems || _ckItems.length === 0) { document.getElementById('ck-progreso').style.display = 'none'; cont.innerHTML = '<p class="vacio">No hay documentos para mostrar.</p>'; return; }
    ckRenderProgreso();
    var grupos = ckAgruparPorCategoria(_ckItems);
    cont.innerHTML = grupos.map(function (g) {
      var esAut = g.categoria.toUpperCase().indexOf('AUTORIZ') !== -1;
      var titulo = esAut ? ('📁 Documentos del Crédito · ' + g.categoria) : g.categoria;
      var filas = g.items.map(ckHtmlFila).join('');
      return '<div class="ck-cat' + (esAut ? ' ck-cat-credito' : '') + '">' +
        '<button class="ck-cat-header" onclick="ckToggleCat(this)"><span class="ck-cat-arrow">▾</span><span class="ck-cat-nombre">' + esc(titulo) + '</span><span class="ck-cat-count">' + g.items.length + (g.items.length === 1 ? ' doc' : ' docs') + '</span></button>' +
        '<div class="ck-cat-body"><table class="ck-tabla">' +

  function ckClaseEstado(estado) {
    switch (String(estado).toLowerCase()) {
      case 'entregado': return 'est-entregado';

  function ckEstaVencido(item) {
    if (!item.fechaVencimiento) return false;
    if (String(item.estado).toLowerCase() === 'no aplica') return false;
    var venc = ckParseFecha(item.fechaVencimiento);
    if (!venc) return false;
    var hoy = new Date(); hoy.setHours(0, 0, 0, 0);
    return venc < hoy;
  }
  function ckEstadoEfectivo(item) { if (ckEstaVencido(item)) return 'Vencido'; return item.estado || 'Pendiente'; }
  function ckParseFecha(iso) {
    var p = String(iso).split('-');
    if (p.length !== 3) return null;
    var y = Number(p[0]), m = Number(p[1]), d = Number(p[2]);
    if (!y || !m || !d) return null;
    var dt = new Date(y, m - 1, d); dt.setHours(0, 0, 0, 0); return dt;
  }
  function ckFechaISO(d) { var mm = ('0' + (d.getMonth() + 1)).slice(-2), dd = ('0' + d.getDate()).slice(-2); return d.getFullYear() + '-' + mm + '-' + dd; }
  function ckHoyISO() { return ckFechaISO(new Date()); }
  function ckSumarMeses(iso, meses) {
    var d = ckParseFecha(iso);
    if (!d) return '';
    var dia = d.getDate();
    d.setDate(1); d.setMonth(d.getMonth() + meses);
    var maxDia = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
    d.setDate(Math.min(dia, maxDia));
    return ckFechaISO(d);
  }
  function ckFechaDocumento(item) {

  }

  function ctrlSeccionVencimientos() {
    var proximo = ctrlVal('Próximo Pago Fecha');
    var vencFinal = ctrlVal('Fecha Vencimiento Final');
    var cards = [];
    var dProx = ctrlDiasHasta(proximo), proxClase, proxTxt;
    if (!proximo) { proxClase = 'ctrl-alert-gris'; proxTxt = 'Sin fecha registrada'; }
    else if (dProx < 0) { proxClase = 'ctrl-alert-rojo'; proxTxt = fechaDisplay(proximo) + ' (vencido hace ' + Math.abs(dProx) + ' día(s))'; }
    else if (dProx <= 7) { proxClase = 'ctrl-alert-ambar'; proxTxt = fechaDisplay(proximo) + ' (en ' + dProx + ' día(s))'; }
    else { proxClase = 'ctrl-alert-verde'; proxTxt = fechaDisplay(proximo) + ' (en ' + dProx + ' día(s))'; }
    cards.push('<div class="ctrl-alert ' + proxClase + '"><span class="ctrl-alert-lbl">Próximo pago</span><span class="ctrl-alert-val">' + esc(proxTxt) + '</span></div>');
    var dFin = ctrlDiasHasta(vencFinal), finClase, finTxt;
    if (!vencFinal) { finClase = 'ctrl-alert-gris'; finTxt = 'Sin fecha registrada'; }
    else if (dFin < 0) { finClase = 'ctrl-alert-rojo'; finTxt = fechaDisplay(vencFinal) + ' (vencido)'; }
    else if (dFin <= 30) { finClase = 'ctrl-alert-ambar'; finTxt = fechaDisplay(vencFinal) + ' (en ' + dFin + ' día(s))'; }
    else { finClase = 'ctrl-alert-verde'; finTxt = fechaDisplay(vencFinal) + ' (en ' + dFin + ' día(s))'; }
    cards.push('<div class="ctrl-alert ' + finClase + '"><span class="ctrl-alert-lbl">Vencimiento final</span><span class="ctrl-alert-val">' + esc(finTxt) + '</span></div>');
    var cuerpo = '<div class="ctrl-alerts">' + cards.join('') + '</div><p class="ctrl-nota">Estas fechas se editan en "Condiciones" y "Estatus".</p>';
    return ctrlCatHTML('Vencimientos', '⏰', cuerpo, false);
  }

  function siacRenderTabla(data, cols, sortState, sortFnName, rowClickFn) {
    if (!data || data.length === 0) return '<p class="vacio">No hay datos en SIAC_Data. Pega el reporte en la hoja "SIAC_Import" y usa "📥 Importar SIAC".</p>';
    var key = sortState.key, dir = sortState.dir;
    var col = cols.filter(function (c) { return c.key === key; })[0] || cols[0];
    var datos = data.slice().sort(function (a, b) {
      var va = a[key], vb = b[key];
      if (col.tipo === 'moneda' || col.tipo === 'pct' || col.tipo === 'num') { va = Number(va) || 0; vb = Number(vb) || 0; return (va - vb) * dir; }
      va = String(va == null ? '' : va).toLowerCase(); vb = String(vb == null ? '' : vb).toLowerCase();
      if (va < vb) return -1 * dir; if (va > vb) return 1 * dir; return 0;
    });
    var ths = cols.map(function (c) {
      var flecha = (sortState.key === c.key) ? '<span class="siac-sort">' + (sortState.dir > 0 ? '▲' : '▼') + '</span>' : '';
      return '<th onclick="' + sortFnName + '(\'' + c.key + '\')">' + esc(c.label) + flecha + '</th>';
    }).join('');
    var filas = datos.map(function (r) {
      var onclick = rowClickFn ? ' onclick="' + rowClickFn + '(\'' + esc(r.idCredito || '') + '\')"' : '';
      return '<tr' + onclick + '>' + cols.map(function (c) { return siacCelda(r, c); }).join('') + '</tr>';
    }).join('');
    return '<div class="siac-wrap"><table class="siac-tabla"><thead><tr>' + ths + '</tr></thead><tbody>' + filas + '</tbody></table></div>';
  }

    var actual = sel.value; opcionesClientes('fmt-cliente'); if (actual) sel.value = actual;
    google.script.run.withSuccessHandler(function (pls) {
      _fmtPlantillas = pls || [];
      var s = document.getElementById('fmt-plantilla');
      s.innerHTML = _fmtPlantillas.length === 0
        ? '<option value="">(Sin plantillas registradas)</option>'
        : '<option value="">Selecciona…</option>' + _fmtPlantillas.map(function (p) { return '<option value="' + esc(p.id) + '">' + esc(p.nombre) + '</option>'; }).join('');
    }).withFailureHandler(function (err) { toast('Error: ' + err.message, true); }).obtenerPlantillas();
  }
  function fmtSeleccionarCliente() {
    _fmtClienteId = document.getElementById('fmt-cliente').value;
    var selC = document.getElementById('fmt-credito');
    selC.value = '';
    if (!_fmtClienteId) { selC.disabled = true; selC.innerHTML = '<option value="">Selecciona un cliente primero…</option>'; return; }
    selC.disabled = true; selC.innerHTML = '<option value="">Cargando…</option>';
    google.script.run.withSuccessHandler(function (creditos) {
      selC.innerHTML = '<option value="">(Sin crédito)</option>' + (creditos || []).map(function (c) { return '<option value="' + esc(c['ID']) + '">' + esc(c['Nombre del crédito']) + '</option>'; }).join('');
      selC.disabled = false;
    }).withFailureHandler(function (err) { selC.innerHTML = '<option value="">Error</option>'; toast('Error: ' + err.message, true); })
      .obtenerCreditosPorCliente(_fmtClienteId, true);
  }
  function fmtAbrirDatosCliente() {
    if (!_fmtClienteId) { toast('Selecciona un cliente primero.', true); return; }
    google.script.run.withSuccessHandler(function (d) {
      document.getElementById('dc-rfc').value = d['RFC'] || '';
      document.getElementById('dc-direccion').value = d['Dirección'] || '';
      document.getElementById('dc-replegal').value = d['Representante Legal'] || '';
      document.getElementById('dc-telefono').value = d['Teléfono'] || '';
      document.getElementById('dc-correo').value = d['Correo'] || '';
      abrirModal('modal-datos-cliente');
    }).withFailureHandler(function (err) { toast('Error: ' + err.message, true); }).obtenerDatosCliente(_fmtClienteId);
  }
  function fmtGuardarDatosCliente() {
    var btn = document.getElementById('btn-guardar-dc');
    btn.disabled = true; btn.textContent = 'Guardando…';
    google.script.run.withSuccessHandler(function () {
      btn.disabled = false; btn.textContent = 'Guardar';
      cerrarModal('modal-datos-cliente'); toast('Datos del cliente guardados.');
    }).withFailureHandler(function (err) { btn.disabled = false; btn.textContent = 'Guardar'; toast('Error: ' + err.message, true); })
      .guardarDatosCliente(_fmtClienteId, {
        'RFC': document.getElementById('dc-rfc').value.trim(),
        'Dirección': document.getElementById('dc-direccion').value.trim(),
        'Representante Legal': document.getElementById('dc-replegal').value.trim(),
        'Teléfono': document.getElementById('dc-telefono').value.trim(),
        'Correo': document.getElementById('dc-correo').value.trim()
      });
  }
  function fmtGenerar() {
    var idPl = document.getElementById('fmt-plantilla').value;
    if (!_fmtClienteId) { toast('Selecciona un cliente.', true); return; }
    if (!idPl) { toast('Selecciona un formato.', true); return; }
    var btn = document.getElementById('btn-fmt-generar');
    btn.disabled = true; btn.textContent = 'Preparando…';
    google.script.run.withSuccessHandler(function (info) {
      btn.disabled = false; btn.textContent = '📄 Generar PDF';
      var idCred = document.getElementById('fmt-credito').value;
      if (info.requiereCredito && !idCred) { toast('Este formato requiere seleccionar un crédito.', true); return; }
      if (info.campos && info.campos.length > 0) {
        _fmtCamposPend = info.campos;
        document.getElementById('fmt-inputs-cuerpo').innerHTML = info.campos.map(function (c) {
          return '<label>' + esc(c.etiqueta) + '<input type="text" id="fmtin-' + esc(c.clave) + '"></label>';
        }).join('');
        abrirModal('modal-fmt-inputs');
      } else { _fmtCamposPend = []; fmtGenerarConInputs(); }
    }).withFailureHandler(function (err) { btn.disabled = false; btn.textContent = '📄 Generar PDF'; toast('Error: ' + err.message, true); })
      .obtenerCamposPlantilla(idPl);
  }
  function fmtGenerarConInputs() {
    var inputs = {};
    (_fmtCamposPend || []).forEach(function (c) {
      var el = document.getElementById('fmtin-' + c.clave);
      inputs[c.clave] = el ? el.value.trim() : '';
    });
    cerrarModal('modal-fmt-inputs');
    var btn = document.getElementById('btn-fmt-generar');
    btn.disabled = true; btn.textContent = 'Generando PDF…';
    toast('Generando formato…');
    google.script.run.withSuccessHandler(function (res) {
      btn.disabled = false; btn.textContent = '📄 Generar PDF';
      var bin = atob(res.base64), bytes = new Uint8Array(bin.length);
      for (var i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
      var blob = new Blob([bytes], { type: 'application/pdf' });
      var url = URL.createObjectURL(blob);
      var a = document.createElement('a');
      a.href = url; a.download = res.nombre || 'Formato.pdf';
      document.body.appendChild(a); a.click(); document.body.removeChild(a);
      setTimeout(function () { URL.revokeObjectURL(url); }, 1500);
      toast('Formato descargado.');
    }).withFailureHandler(function (err) { btn.disabled = false; btn.textContent = '📄 Generar PDF'; toast('Error: ' + err.message, true); })
      .generarFormato(document.getElementById('fmt-plantilla').value, _fmtClienteId,
        document.getElementById('fmt-credito').value || null, inputs);
  }
