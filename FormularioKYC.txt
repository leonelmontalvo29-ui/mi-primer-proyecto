/* ==========================================
   PERSONA FÍSICA / MORAL
========================================== */

function cambiarTipoPersona() {

  const tipo =
    document.getElementById('tipoPersona').value;

  const pf =
    document.getElementById('formPF');

  const aviso =
    document.getElementById('avisoPM');


  if (tipo === 'PF') {

    pf.classList.remove('oculto');
    aviso.classList.add('oculto');

  } else if (tipo === 'PM') {

    pf.classList.add('oculto');
    aviso.classList.remove('oculto');

  } else {

    pf.classList.add('oculto');
    aviso.classList.add('oculto');

  }
}



/* ==========================================
   RESIDENCIA EN MÉXICO
========================================== */

function evaluarResidencia() {

  const pais =
    document.getElementById('paisResidencia')
      .value
      .trim()
      .toLowerCase();

  const contenedor =
    document.getElementById(
      'motivoMexicoContenedor'
    );

  if (
    pais &&
    pais !== 'mexico' &&
    pais !== 'méxico'
  ) {

    contenedor.classList.remove('oculto');

  } else {

    contenedor.classList.add('oculto');

  }

}

function iniciarVinculacionKYC() {

  google.script.run

    .withSuccessHandler(function(clientes) {

      const select =
        document.getElementById('kycCliente');

      select.innerHTML =
        '<option value="">Seleccione un cliente</option>';

      clientes.forEach(function(cliente) {

        const option =
          document.createElement('option');

        option.value = cliente.id;
        option.textContent =
          cliente.nombre + ' (' + cliente.id + ')';

        select.appendChild(option);
      });

    })

    .withFailureHandler(function(error) {

      alert(
        'No fue posible cargar los clientes:\n' +
        error.message
      );

    })

    .obtenerClientesKYC();
}


function cargarCreditosKYC() {

  const idCliente =
    document.getElementById('kycCliente').value;

  const credito =
    document.getElementById('kycCredito');

  const personalidad =
    document.getElementById('kycPersonalidad');


  credito.innerHTML =
    '<option value="">Cargando...</option>';

  personalidad.innerHTML =
    '<option value="">Primero seleccione un crédito</option>';

  personalidad.disabled = true;


  if (!idCliente) {

    credito.innerHTML =
      '<option value="">Primero seleccione un cliente</option>';

    credito.disabled = true;

    return;
  }


  google.script.run

    .withSuccessHandler(function(creditos) {

      credito.innerHTML =
        '<option value="">Seleccione un crédito</option>';


      creditos.forEach(function(item) {

        const option =
          document.createElement('option');

        option.value = item.id;

        option.textContent =
          item.nombre + ' (' + item.id + ')';

        credito.appendChild(option);
      });


      credito.disabled = false;


      if (creditos.length === 0) {

        credito.innerHTML =
          '<option value="">Este cliente no tiene créditos</option>';

        credito.disabled = true;
      }

    })

    .withFailureHandler(function(error) {

      alert(
        'No fue posible cargar los créditos:\n' +
        error.message
      );

    })

    .obtenerCreditosKYC(idCliente);
}


function cargarPersonalidadesKYC() {

  const idCredito =
    document.getElementById('kycCredito').value;

  const select =
    document.getElementById('kycPersonalidad');


  if (!idCredito) {

    select.innerHTML =
      '<option value="">Primero seleccione un crédito</option>';

    select.disabled = true;

    return;
  }


  select.innerHTML =
    '<option value="">Cargando...</option>';


  google.script.run

    .withSuccessHandler(function(personalidades) {

      select.innerHTML =
        '<option value="">Seleccione una personalidad</option>';


      personalidades.forEach(function(persona) {

        const option =
          document.createElement('option');

        option.value = persona.id;

        option.textContent =
          persona.nombre +
          (persona.tipo ? ' — ' + persona.tipo : '');

        select.appendChild(option);
      });


      select.disabled = false;


      if (personalidades.length === 0) {

        select.innerHTML =
          '<option value="">Este crédito no tiene personalidades</option>';

        select.disabled = true;
      }

    })

    .withFailureHandler(function(error) {

      alert(
        'No fue posible cargar las personalidades:\n' +
        error.message
      );

    })

    .obtenerPersonalidadesKYC(idCredito);
}
